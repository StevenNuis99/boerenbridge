function renderGameDetail(potje) { const container = document.getElementById("game-detail"); const rows = potje.resultaten .sort((a, b) => a.plaats - b.plaats) .map(r => ` ${potje.gameId} ${r.speler} - - ${r.slagenKoning ? 'Ja' : 'Nee'} ${r.plaats} ${r.cpTotaal.toFixed(1)} `) .join(""); container.innerHTML = `
📊 Details van potje ${potje.gameId}
${rows}
Game-ID Speler Punten Bonus Slagen koning Rank CP totaal
`; }
